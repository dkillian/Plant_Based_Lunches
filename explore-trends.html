Data Loading

The State of Origin reports changed format across years. Three loaders handle the three distinct schemas (FY18–21, FY22–23, FY24–25), then all eight files are harmonised into a single tidy table.

Code 
# FY18–FY21: 7 cols, 1-row title header to skip
load_soo_early <- function(path, year) {
  read_excel(path, skip = 1) |>
    select(
      origin_state  = 1, material_group = 3, material_code = 4,
      material_desc = 5, qty_lbs = 6,        value_usd = 7
    ) |>
    mutate(
      fiscal_year   = year,
      material_code = as.character(material_code),
      qty_lbs       = as.numeric(qty_lbs),
      value_usd     = as.numeric(value_usd)
    )
}

# FY22–FY23: 6 cols, no skip
load_soo_mid <- function(path, year) {
  read_excel(path) |>
    select(
      origin_state  = 1, material_group = 2, material_code = 3,
      material_desc = 4, qty_lbs = 5,        value_usd = 6
    ) |>
    mutate(
      fiscal_year   = year,
      material_code = as.character(material_code),
      qty_lbs       = as.numeric(qty_lbs),
      value_usd     = as.numeric(value_usd)
    )
}

# FY24–FY25: hierarchical format; state rows identified by NA material_code
load_soo_late <- function(path, year) {
  df <- read_excel(path)
  names(df) <- c("state_or_product", "material_code", "qty_lbs", "value_usd")
  df |>
    mutate(
      origin_state = if_else(is.na(material_code), state_or_product, NA_character_)
    ) |>
    fill(origin_state, .direction = "down") |>
    filter(!is.na(material_code)) |>
    mutate(
      material_desc  = state_or_product,
      material_group = NA_character_,
      fiscal_year    = year,
      material_code  = as.character(material_code),
      qty_lbs        = as.numeric(qty_lbs),
      value_usd      = as.numeric(value_usd)
    ) |>
    select(origin_state, material_group, material_code,
           material_desc, qty_lbs, value_usd, fiscal_year)
}

soo_raw <- bind_rows(
  load_soo_early("data/FY18 State of Origin CNP- FINAL.xlsx",                  2018),
  load_soo_early("data/FY19 State of Origin CNP- FINAL.xlsx",                  2019),
  load_soo_early("data/FY20 SOO CNP Only Data_filtered and formatted.xlsx",    2020),
  load_soo_early("data/FY21SOOCNPProgramOnlyData.xls.xlsx",                    2021),
  load_soo_mid(  "data/FY22StateofOrigin-CNP.xlsx",                            2022),
  load_soo_mid(  "data/FY23StateofOrigin-CNP.xlsx",                            2023),
  load_soo_late( "data/FY24StateofOriginReport-CNP.xlsx",                      2024),
  load_soo_late( "data/FY25StateofOriginReport-CNP.xlsx",                      2025)
)

# Remove subtotal / grand total rows
soo <- soo_raw |>
  filter(
    !str_detect(origin_state, regex("total", ignore_case = TRUE)),
    !is.na(material_code)
  )

After cleaning, the dataset contains 11,953 purchase records across eight fiscal years.

Note

FY2024 and FY2025 source files use a condensed format that drops the material group (food category) column. Trend charts that rely on that column are therefore limited to FY2018–FY2023.

Animal Protein Share

The material group field is absent from FY2024–FY2025 source files, but material codes are consistent across all years. A code-to-group lookup built from FY2018–FY2023 — supplemented with 11 manually assigned codes for new FY2024–FY2025 items — allows the classification to extend across the full period.

code_lookup <- soo |>
  filter(!is.na(material_group), !is.na(material_code)) |>
  distinct(material_code, material_group)

soo_full <- soo |>
  left_join(full_lookup |> rename(group_filled = material_group),
            by = "material_code") |>
  mutate(material_group = coalesce(material_group, group_filled)) |>
  select(-group_filled)

soo_full |>
  filter(!is.na(material_group)) |>
  group_by(fiscal_year) |>
  summarise(
    total     = sum(value_usd, na.rm = TRUE),
    meat_val  = sum(value_usd[material_group %in% meat_groups],  na.rm = TRUE),
    dairy_val = sum(value_usd[material_group %in% dairy_groups], na.rm = TRUE),
    egg_val   = sum(value_usd[material_group %in% egg_groups],   na.rm = TRUE),
    plant_val = total - meat_val - dairy_val - egg_val,
    .groups   = "drop"
  ) |>
  pivot_longer(c(meat_val, dairy_val, egg_val, plant_val),
               names_to = "type", values_to = "value") |>
  mutate(
    share = value / total,
    type  = factor(type,
                   levels = c("plant_val", "egg_val", "dairy_val", "meat_val"),
                   labels = c("Plant-based", "Eggs", "Dairy", "Meat/Poultry/Fish"))
  ) |>
  ggplot(aes(x = fiscal_year, y = share, fill = type)) +
  geom_area(position = "stack") +
  scale_x_continuous(breaks = 2018:2025) +
  scale_y_continuous(labels = label_percent(), expand = c(0, 0)) +
  scale_fill_manual(
    values = c(
      "Meat/Poultry/Fish" = "#8B2020",
      "Dairy"             = "#D4956A",
      "Eggs"              = "#F2C96E",
      "Plant-based"       = "#5A8A5A"
    )
  ) +
  labs(
    x    = "Fiscal Year",
    y    = NULL,
    fill = NULL,
    title = "Share of USDA CNP Purchases by Protein Type",
    subtitle = "Dollar value, FY2018–FY2025"
  ) +
  theme(legend.position = "bottom")
